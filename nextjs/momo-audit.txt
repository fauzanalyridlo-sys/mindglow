"use client";

import {
  useCallback,
  useEffect,
  useRef,
  useState,
  useSyncExternalStore,
} from "react";
import "./momo-pet.css";
import { CatSprite, CreeperSprite, type EyeState } from "./MomoSprites";
import { useWorld, type Phase, type Weather } from "./momo-world";
import {
  ACCESSORIES,
  DEFAULT_SAVED,
  SKINS,
  applyOffline,
  clamp,
  greeting,
  levelProgress,
  unlocksAt,
  type Saved,
} from "./momo-data";

type Action = "idle" | "happy" | "eat" | "play" | "puff";
type Particle = { id: number; emoji: string; dx: number };
type Orb = { id: number; x: number; y: number; born: number };
type Burst = { id: number; x: number; y: number };
type Creeper = { x: number; dir: 1 | -1; fleeing: boolean };

const MIN_X = 18;
const MAX_X = 82;

const FEED_LINES = [
  "Nyam nyam! Ikannya enak! 🐟",
  "Ikan mentah favoritku! 🐟",
  "Terima kasih! Perutku hangat 💗",
];

/* Apakah tab sedang terlihat (timer dijeda kalau tersembunyi) */
function subscribeVisibility(cb: () => void) {
  document.addEventListener("visibilitychange", cb);
  return () => document.removeEventListener("visibilitychange", cb);
}

function usePageVisible() {
  return useSyncExternalStore(
    subscribeVisibility,
    () => document.visibilityState !== "hidden",
    () => true,
  );
}

export default function MomoPet() {
  const world = useWorld();
  const visible = usePageVisible();

  /* ---------- data utama (tersimpan) ---------- */
  const [pet, setPet] = useState<Saved>(DEFAULT_SAVED);
  const [hydrated, setHydrated] = useState(false);
  const [userId, setUserId] = useState<string | null>(null);
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [mobileCollapsed, setMobileCollapsed] = useState(true);
  const [confirmReset, setConfirmReset] = useState(false);

  /* ---------- state sementara ---------- */
  const [message, setMessage] = useState<string | null>(null);
  const [action, setAction] = useState<Action>("idle");
  const [x, setX] = useState(50);
  const [facing, setFacing] = useState<1 | -1>(1);
  const [walking, setWalking] = useState(false);
  const [dragging, setDragging] = useState(false);
  const [particles, setParticles] = useState<Particle[]>([]);
  const [bursts, setBursts] = useState<Burst[]>([]);
  const [orbs, setOrbs] = useState<Orb[]>([]);
  const [creeper, setCreeper] = useState<Creeper | null>(null);
  const [fishKey, setFishKey] = useState(0);

  /* ---------- refs ---------- */
  const stageRef = useRef<HTMLDivElement>(null);
  const dragRef = useRef({
    active: false,
    moved: false,
    startX: 0,
  });
  const petRef = useRef(pet);
  const xRef = useRef(50);
  const cursorRef = useRef<{ x: number; t: number } | null>(null);
  const creeperRef = useRef<Creeper | null>(null);
  const walkTimer = useRef<number | null>(null);
  const fishTimer = useRef<number | null>(null);
  const particleId = useRef(0);
  const orbId = useRef(0);
  const burstId = useRef(0);
  const wasSleeping = useRef(false);
  const prevLevel = useRef(1);
  const prevWeather = useRef<Weather>("clear");
  const prevPhase = useRef<Phase>("day");
  const worldInit = useRef(false);
  const pendingXpRef = useRef<Set<Promise<void>>>(new Set());
  const scoreRevisionRef = useRef(0);
  const resettingRef = useRef(false);

  useEffect(() => {
    petRef.current = pet;
  }, [pet]);

  useEffect(() => {
    xRef.current = x;
  }, [x]);

  const { level, pct } = levelProgress(pet.score);
  const sleeping = pet.asleep;
  const skin = SKINS.find((s) => s.id === pet.skin) ?? SKINS[0];
  const displayName = pet.name.trim() || "Momo";
  const eyes: EyeState = sleeping
    ? "sleep"
    : action === "happy" || action === "play"
      ? "happy"
      : "open";

  /* ---------- helpers ---------- */
  const say = useCallback((text: string) => setMessage(text), []);

  const awardXp = useCallback((
    activity: "creeper" | "pet" | "feed" | "play" | "orb"
  ): Promise<void> => {
    if (resettingRef.current) {
      return Promise.resolve();
    }

    const revision = scoreRevisionRef.current;
    const requestId = crypto.randomUUID();

    const request = (async () => {
      try {
        const response = await fetch("/api/momo/xp", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            action: activity,
            requestId,
          }),
        });

        const result = (await response.json()) as {
          success?: boolean;
          message?: string;
          data?: {
            score?: number;
            awarded?: boolean;
          };
        };

        if (
          !response.ok ||
          !result.success ||
          typeof result.data?.score !== "number"
        ) {
          throw new Error(result.message ?? "Gagal menyimpan XP Momo.");
        }

        const serverScore = result.data.score;

        if (revision === scoreRevisionRef.current) {
          setPet((current) => ({
            ...current,
            score: Math.max(current.score, serverScore),
          }));
        }
      } catch (error) {
        console.error("Gagal menambahkan XP Momo:", error);

        if (revision === scoreRevisionRef.current) {
          setMessage("XP Momo gagal disimpan. Periksa koneksi internet.");
        }
      }
    })();

    pendingXpRef.current.add(request);

    void request.finally(() => {
      pendingXpRef.current.delete(request);
    });

    return request;
  }, []);

  const spawn = useCallback((emoji: string, count = 3) => {
    const items: Particle[] = Array.from({ length: count }, () => ({
      id: ++particleId.current,
      emoji,
      dx: Math.round(Math.random() * 60 - 30),
    }));

    setParticles((p) => [...p, ...items]);

    window.setTimeout(() => {
      setParticles((p) => p.filter((i) => !items.some((n) => n.id === i.id)));
    }, 1200);
  }, []);

  const moveTo = useCallback((target: number) => {
    setFacing(target > xRef.current ? 1 : -1);
    setWalking(true);
    setX(target);

    if (walkTimer.current) {
      window.clearTimeout(walkTimer.current);
    }

    walkTimer.current = window.setTimeout(() => setWalking(false), 1700);
  }, []);

  /* ---------- muat data Momo dari database ---------- */
  useEffect(() => {
    let cancelled = false;

    async function initializePet() {
      try {
        const response = await fetch("/api/momo", {
          method: "GET",
          cache: "no-store",
        });

        if (!response.ok) {
          throw new Error(`Gagal memuat Momo: HTTP ${response.status}`);
        }

        const result = (await response.json()) as {
          success?: boolean;
          isNew?: boolean;
          userId?: string;
          data?: {
            name?: unknown;
            score?: unknown;
            energy?: unknown;
            happiness?: unknown;
            fullness?: unknown;
            asleep?: unknown;
            skin?: unknown;
            accessory?: unknown;
            animations?: unknown;
            autoMove?: unknown;
            chase?: unknown;
            lastSeen?: unknown;
            lastCheckin?: unknown;
          };
        };

        if (
          !result.success ||
          !result.data ||
          typeof result.userId !== "string" ||
          !result.userId
        ) {
          throw new Error("Identitas pengguna atau data Momo tidak valid.");
        }

        if (cancelled) return;

        const activeUserId = result.userId;
        const remote = result.data;
        const now = Date.now();

        const safeNumber = (value: unknown, fallback: number) =>
          typeof value === "number" && Number.isFinite(value)
            ? value
            : fallback;

        const databaseSaved: Saved = {
          ...DEFAULT_SAVED,
          name:
            typeof remote.name === "string" && remote.name.trim()
              ? remote.name.trim().slice(0, 12)
              : DEFAULT_SAVED.name,
          score: Math.max(
            0,
            Math.min(1_000_000, Math.floor(safeNumber(remote.score, 0)))
          ),
          energy: clamp(safeNumber(remote.energy, DEFAULT_SAVED.energy)),
          happiness: clamp(
            safeNumber(remote.happiness, DEFAULT_SAVED.happiness)
          ),
          fullness: clamp(
            safeNumber(remote.fullness, DEFAULT_SAVED.fullness)
          ),
          asleep:
            typeof remote.asleep === "boolean"
              ? remote.asleep
              : DEFAULT_SAVED.asleep,
          skin: SKINS.some((item) => item.id === remote.skin)
            ? (remote.skin as Saved["skin"])
            : DEFAULT_SAVED.skin,
          accessory: ACCESSORIES.some(
            (item) => item.id === remote.accessory
          )
            ? (remote.accessory as Saved["accessory"])
            : DEFAULT_SAVED.accessory,
          animations:
            typeof remote.animations === "boolean"
              ? remote.animations
              : DEFAULT_SAVED.animations,
          autoMove:
            typeof remote.autoMove === "boolean"
              ? remote.autoMove
              : DEFAULT_SAVED.autoMove,
          chase:
            typeof remote.chase === "boolean"
              ? remote.chase
              : DEFAULT_SAVED.chase,
          lastSeen: safeNumber(remote.lastSeen, now),
          lastCheckin:
            typeof remote.lastCheckin === "string"
              ? remote.lastCheckin
              : "",
        };

        // Terapkan penurunan stat berdasarkan waktu terakhir tersimpan di DB.
        const offline = applyOffline(databaseSaved, now);
        const loaded: Saved = {
          ...offline.saved,
          lastSeen: now,
        };

        const currentLevel = levelProgress(loaded.score).level;
        const selectedSkin = SKINS.find((item) => item.id === loaded.skin);
        const selectedAccessory = ACCESSORIES.find(
          (item) => item.id === loaded.accessory
        );

        if (!selectedSkin || selectedSkin.level > currentLevel) {
          loaded.skin = "salju";
        }

        if (
          !selectedAccessory ||
          selectedAccessory.level > currentLevel
        ) {
          loaded.accessory = "none";
        }

        if (cancelled) return;

        setUserId(activeUserId);
        prevLevel.current = levelProgress(loaded.score).level;
        wasSleeping.current = loaded.asleep;
        petRef.current = loaded;

        setPet(loaded);
        setHydrated(true);

        setMessage(
          greeting(
            offline.hours,
            new Date().getHours(),
            Boolean(result.isNew),
            loaded.name,
          )
        );
      } catch (error) {
        if (!cancelled) {
          console.error("Gagal memuat data Momo:", error);
          setMessage("Data Momo gagal dimuat. Silakan muat ulang halaman.");
        }
      }
    }

    void initializePet();

    return () => {
      cancelled = true;
    };
  }, [spawn]);

  /* ---------- simpan seluruh data Momo ke database ---------- */
  useEffect(() => {
    if (!hydrated || !userId) return;

    const saved: Saved = {
      ...pet,
      lastSeen: Date.now(),
    };

    petRef.current = saved;

    const timer = window.setTimeout(() => {
      void fetch("/api/momo", {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          name: saved.name,
          energy: saved.energy,
          happiness: saved.happiness,
          fullness: saved.fullness,
          asleep: saved.asleep,
          skin: saved.skin,
          accessory: saved.accessory,
          animations: saved.animations,
          autoMove: saved.autoMove,
          chase: saved.chase,
        }),
      })
        .then((response) => {
          if (!response.ok) {
            console.error("Gagal menyimpan data Momo ke database.");
          }
        })
        .catch((error) => {
          console.error("Gagal menyinkronkan data Momo:", error);
        });
    }, 700);

    return () => window.clearTimeout(timer);
  }, [pet, hydrated, userId]);

  /* ---------- hitung waktu saat tab tidak aktif ---------- */
  useEffect(() => {
    if (!hydrated || !userId) return;

    let hiddenAt = 0;

    function onVisibility() {
      if (document.visibilityState === "hidden") {
        hiddenAt = Date.now();

        // Kirim status terakhir langsung ke database saat tab disembunyikan.
        const saved = petRef.current;

        void fetch("/api/momo", {
          method: "PUT",
          headers: { "Content-Type": "application/json" },
          keepalive: true,
          body: JSON.stringify({
            name: saved.name,
            energy: saved.energy,
            happiness: saved.happiness,
            fullness: saved.fullness,
            asleep: saved.asleep,
            skin: saved.skin,
            accessory: saved.accessory,
            animations: saved.animations,
            autoMove: saved.autoMove,
            chase: saved.chase,
          }),
        }).catch((error) => {
          console.error("Gagal menyimpan Momo saat tab disembunyikan:", error);
        });
      } else if (hiddenAt) {
        const now = Date.now();

        if (now - hiddenAt > 5 * 60_000) {
          const { saved, hours } = applyOffline(
            {
              ...petRef.current,
              lastSeen: hiddenAt,
            },
            now
          );

          const resumed: Saved = {
            ...saved,
            lastSeen: now,
          };

          petRef.current = resumed;
          setPet(resumed);

          setMessage(
            hours >= 6
              ? "Kamu lama tidak datang... aku kangen! 💗"
              : "Selamat datang kembali! 👋"
          );
        }

        hiddenAt = 0;
      }
    }

    document.addEventListener("visibilitychange", onVisibility);

    return () => {
      document.removeEventListener("visibilitychange", onVisibility);
    };
  }, [hydrated, userId]);

  /* ---------- reset action ---------- */
  useEffect(() => {
    if (action === "idle") return;

    const timer = window.setTimeout(
      () => setAction("idle"),
      action === "puff" ? 1300 : 1000,
    );

    return () => window.clearTimeout(timer);
  }, [action]);

  /* ---------- sembunyikan bubble ---------- */
  useEffect(() => {
    if (!message) return;

    const timer = window.setTimeout(() => setMessage(null), 4500);

    return () => window.clearTimeout(timer);
  }, [message]);

  /* ---------- stat turun pelan setiap 30 detik ---------- */
  useEffect(() => {
    if (!hydrated || !visible) return;

    const weather = world.weather;

    const timer = window.setInterval(() => {
      const now = Date.now();

      setPet((p) => {
        const energy = p.asleep ? clamp(p.energy + 50) : clamp(p.energy - 1);

        const asleep = p.asleep ? energy < 50 : energy < 20;

        const moodShift = weather === "rain" ? -2 : weather === "snow" ? 0 : -1;

        return {
          ...p,
          fullness: clamp(p.fullness - 1),
          happiness: clamp(p.happiness + moodShift),
          energy,
          asleep,
          lastSeen: now,
        };
      });
    }, 30_000);

    return () => window.clearInterval(timer);
  }, [hydrated, visible, world.weather]);

  /* ---------- pesan tidur / bangun ---------- */
  useEffect(() => {
    if (!hydrated) return;

    const previousSleeping = wasSleeping.current;
    wasSleeping.current = sleeping;

    if (previousSleeping === sleeping) return;

    const timer = window.setTimeout(
      () =>
        say(
          sleeping
            ? "Zzz... aku ngantuk 😴"
            : "Hoaam... aku sudah segar lagi! ☀️",
        ),
      0,
    );

    return () => window.clearTimeout(timer);
  }, [sleeping, hydrated, say]);

  /* ---------- pesan saat cuaca / malam berubah ---------- */
  useEffect(() => {
    if (!hydrated) return;

    if (!worldInit.current) {
      worldInit.current = true;
      prevWeather.current = world.weather;
      prevPhase.current = world.phase;
      return;
    }

    const previousWeather = prevWeather.current;
    const previousPhase = prevPhase.current;

    prevWeather.current = world.weather;
    prevPhase.current = world.phase;

    let text: string | null = null;

    if (world.weather !== previousWeather) {
      text =
        world.weather === "rain"
          ? "Hujan turun... 🌧️ aku jadi agak murung"
          : world.weather === "snow"
            ? "Salju! ❄️ Dingin tapi seru!"
            : "Cuacanya cerah lagi! ☀️";
    } else if (world.phase !== previousPhase && world.phase === "night") {
      text = "Sudah malam. Bintangnya cantik ya ✨";
    }

    if (!text) return;

    const nextMessage = text;
    const timer = window.setTimeout(() => say(nextMessage), 0);

    return () => window.clearTimeout(timer);
  }, [hydrated, world.weather, world.phase, say]);

  /* ---------- pengingat istirahat larut malam ---------- */
  useEffect(() => {
    if (!hydrated || !visible) return;

    const timer = window.setInterval(() => {
      const hour = new Date().getHours();

      if ((hour >= 23 || hour < 4) && !petRef.current.asleep) {
        setMessage("Sudah larut... istirahat dulu ya, besok lanjut lagi 🌙");
      }
    }, 15 * 60_000);

    return () => window.clearInterval(timer);
  }, [hydrated, visible]);

  /* ---------- level naik ---------- */
  useEffect(() => {
    if (!hydrated) return;

    if (level <= prevLevel.current) {
      prevLevel.current = level;
      return;
    }

    prevLevel.current = level;
    const items = unlocksAt(level);

    const timer = window.setTimeout(() => {
      setAction("play");
      spawn("⭐", 5);

      say(
        `Level ${level}! ${
          items.length ? `Baru: ${items.join(", ")} 🎁` : "🎉"
        }`,
      );
    }, 0);

    return () => window.clearTimeout(timer);
  }, [level, hydrated, say, spawn]);

  /* ---------- lacak posisi kursor di sekitar area Momo ---------- */
  useEffect(() => {
    function onMove(e: PointerEvent) {
      const el = stageRef.current;
      if (!el) return;

      const rect = el.getBoundingClientRect();

      if (
        e.clientY < rect.top - 60 ||
        e.clientY > rect.bottom + 60 ||
        e.clientX < rect.left - 60 ||
        e.clientX > rect.right + 60
      ) {
        cursorRef.current = null;
        return;
      }

      cursorRef.current = {
        x: clamp(((e.clientX - rect.left) / rect.width) * 100, MIN_X, MAX_X),
        t: performance.now(),
      };
    }

    window.addEventListener("pointermove", onMove, {
      passive: true,
    });

    return () => {
      window.removeEventListener("pointermove", onMove);
    };
  }, []);

  /* ---------- jalan sendiri / mengejar kursor ---------- */
  useEffect(() => {
    if (!hydrated || !visible || pet.asleep || !pet.autoMove) {
      return;
    }

    const chase = pet.chase;

    const timer = window.setInterval(() => {
      if (dragRef.current.active) return;

      const cursor = cursorRef.current;

      if (chase && cursor && performance.now() - cursor.t < 2500) {
        if (Math.abs(cursor.x - xRef.current) < 8) {
          return;
        }

        const step = Math.max(-24, Math.min(24, cursor.x - xRef.current));

        moveTo(clamp(xRef.current + step, MIN_X, MAX_X));

        return;
      }

      if (Math.random() < 0.35) return;

      const target = clamp(
        xRef.current + (Math.random() * 50 - 25),
        MIN_X,
        MAX_X,
      );

      if (Math.abs(target - xRef.current) < 6) return;

      moveTo(target);
    }, 1500);

    return () => window.clearInterval(timer);
  }, [hydrated, visible, pet.asleep, pet.autoMove, pet.chase, moveTo]);

  /* ---------- creeper: muncul sesekali ---------- */
  useEffect(() => {
    if (!hydrated || !visible) return;

    const timer = window.setInterval(() => {
      if (creeperRef.current || petRef.current.asleep || Math.random() > 0.35) {
        return;
      }

      const fromLeft = Math.random() < 0.5;

      const nextCreeper: Creeper = {
        x: fromLeft ? -10 : 110,
        dir: fromLeft ? 1 : -1,
        fleeing: false,
      };

      creeperRef.current = nextCreeper;
      setCreeper(nextCreeper);
    }, 25_000);

    return () => window.clearInterval(timer);
  }, [hydrated, visible]);

  const hasCreeper = creeper !== null;

  useEffect(() => {
    if (!hasCreeper || !visible) return;

    const timer = window.setInterval(() => {
      const current = creeperRef.current;
      if (!current) return;

      let next: Creeper = {
        ...current,
        x: current.x + current.dir * (current.fleeing ? 2.4 : 0.9),
      };

      if (
        !next.fleeing &&
        !petRef.current.asleep &&
        Math.abs(next.x - xRef.current) < 20
      ) {
        next = {
          ...next,
          fleeing: true,
          dir: next.dir === 1 ? -1 : 1,
        };

        setAction("puff");
        setMessage("Hsss! Pergi sana, Creeper! 😾");

        setPet((p) => ({
          ...p,
          happiness: clamp(p.happiness + 3),
        }));

        void awardXp("creeper");
      }

      if (next.x < -16 || next.x > 116) {
        creeperRef.current = null;
        setCreeper(null);
        return;
      }

      creeperRef.current = next;
      setCreeper(next);
    }, 120);

    return () => window.clearInterval(timer);
  }, [hasCreeper, visible]);

  /* ---------- XP orb: muncul acak, hilang sendiri ---------- */
  useEffect(() => {
    if (!hydrated || !visible) return;

    const timer = window.setInterval(() => {
      const now = Date.now();
      const skip = Math.random() > 0.7;

      const fresh: Orb = {
        id: ++orbId.current,
        x: MIN_X - 8 + Math.random() * (MAX_X - MIN_X + 16),
        y: 50 + Math.random() * 100,
        born: now,
      };

      setOrbs((previous) => {
        const alive = previous.filter((orb) => now - orb.born < 12_000);

        return alive.length >= 3 || skip ? alive : [...alive, fresh];
      });
    }, 7000);

    return () => window.clearInterval(timer);
  }, [hydrated, visible]);

  /* ---------- aksi elus Momo ---------- */
  function petMomo() {
    if (sleeping) {
      say(`Zzz... (${displayName} masih tidur`);
      return;
    }

    setPet((p) => ({
      ...p,
      happiness: clamp(p.happiness + 8),
    }));

    void awardXp("pet");

    setAction("happy");
    spawn("💗");
    say("Meow! Aku suka dielus! 💗");
  }

  /* ---------- aksi memberi makan ---------- */
  function feedMomo() {
    if (sleeping) {
      say(`Zzz... (${displayName} masih tidur`);
      return;
    }

    if (pet.fullness >= 100) {
      setAction("happy");
      say("Aku sudah kenyang! 🐟");
      return;
    }

    setPet((p) => ({
      ...p,
      fullness: clamp(p.fullness + 15),
      energy: clamp(p.energy + 10),
    }));

    void awardXp("feed");

    setAction("eat");
    setFishKey((key) => key + 1);

    if (fishTimer.current) {
      window.clearTimeout(fishTimer.current);
    }

    fishTimer.current = window.setTimeout(() => setFishKey(0), 900);

    say(FEED_LINES[Math.floor(Math.random() * FEED_LINES.length)]);
  }

  /* ---------- aksi bermain dengan Momo ---------- */
  function playWithMomo() {
    if (sleeping) {
      say(`Zzz... (${displayName} masih tidur`);
      return;
    }

    if (pet.energy < 15) {
      say("Aku capek. Istirahat dulu, ya!");
      return;
    }

    setPet((p) => ({
      ...p,
      happiness: clamp(p.happiness + 12),
      energy: clamp(p.energy - 15),
      fullness: clamp(p.fullness - 5),
    }));

    void awardXp("play");

    setAction("play");
    spawn("✨", 4);
    say("Yeay! Main lagi! ✨");
  }

  /* ---------- kumpulkan XP orb ---------- */
  function collectOrb(orb: Orb) {
    setOrbs((previous) => previous.filter((item) => item.id !== orb.id));

    setPet((p) => ({
      ...p,
      happiness: clamp(p.happiness + 1),
    }));

    void awardXp("orb");

    const id = ++burstId.current;

    setBursts((previous) => [
      ...previous,
      {

export function applyOffline(s: Saved, now: number): { saved: Saved; hours: number } {
    const hours = Math.max(0, Math.min(48, (now - s.lastSeen) / 3_600_000))
    if (hours < 0.1) return { saved: s, hours }

    const fullness = Math.max(Math.min(s.fullness, 15), Math.round(s.fullness - hours * 4))
    const happiness = Math.max(Math.min(s.happiness, 20), Math.round(s.happiness - hours * 3))
    const energy = Math.min(100, Math.round(s.energy + hours * 8))
    const asleep = energy < 20 || (s.asleep && energy < 50)

    return { hours, saved: { ...s, fullness, happiness, energy, asleep } }
}

import { useSyncExternalStore } from 'react'

export type Phase = 'dawn' | 'day' | 'dusk' | 'night'
export type Weather = 'clear' | 'rain' | 'snow'
export type World = { hour: number; phase: Phase; weather: Weather }

/* Nilai awal server agar hidrasi tidak bentrok */
const SERVER_WORLD: World = {
    hour: 12,
    phase: 'day',
    weather: 'clear',
}

export function phaseOf(hour: number): Phase {
    if (hour >= 5 && hour < 7) return 'dawn'
    if (hour >= 7 && hour < 17) return 'day'
    if (hour >= 17 && hour < 19) return 'dusk'
    return 'night'
}

const CLOCK_CHECK_INTERVAL = 30_000
const WEATHER_CHECK_INTERVAL = 3 * 60 * 1000
const MIN_SNOW_DURATION = 6 * 60 * 1000
const SNOW_CHANCE = 0.02
const CLEAR_CHANCE = 0.35

let world: World = SERVER_WORLD
let naturalWeather: Weather = 'clear'
let weatherStartedAt = Date.now()

let hourOverride: number | null = null
let weatherOverride: Weather | null = null

let timers: number[] = []
const listeners = new Set<() => void>()

function currentHour(): number {
    if (hourOverride !== null) return hourOverride
    return new Date().getHours()
}

function recompute() {
    const hour = Math.floor(currentHour())

    /* Hujan dinonaktifkan: cuaca otomatis hanya cerah atau salju */
    const weather: Weather =
        weatherOverride === 'snow' ? 'snow' : naturalWeather

    const phase = phaseOf(hour)

    if (
        hour === world.hour &&
        weather === world.weather &&
        phase === world.phase
    ) {
        return
    }

    world = { hour, phase, weather }
    listeners.forEach((listener) => listener())
}

function rollWeather() {
    const now = Date.now()
    const elapsed = now - weatherStartedAt

    if (naturalWeather === 'snow') {
        if (elapsed < MIN_SNOW_DURATION) return

        if (Math.random() < CLEAR_CHANCE) {
            naturalWeather = 'clear'
            weatherStartedAt = now
        }
    } else if (Math.random() < SNOW_CHANCE) {
        naturalWeather = 'snow'
        weatherStartedAt = now
    }

    recompute()
}

function start() {
    recompute()

    if (timers.length > 0) return

    timers = [
        window.setInterval(recompute, CLOCK_CHECK_INTERVAL),
        window.setInterval(rollWeather, WEATHER_CHECK_INTERVAL),
    ]
}

function stop() {
    timers.forEach((timer) => window.clearInterval(timer))
    timers = []
}

function subscribe(callback: () => void) {
    listeners.add(callback)

    if (listeners.size === 1) start()

    return () => {
        listeners.delete(callback)

        if (listeners.size === 0) stop()
    }
}

const getSnapshot = () => world
const getServerSnapshot = () => SERVER_WORLD

export function useWorld(): World {
    return useSyncExternalStore(
        subscribe,
        getSnapshot,
        getServerSnapshot,
    )
}

/*
 * Pengujian lewat console browser:
 * window.momoWorld.set({ hour: 22, weather: 'snow' })
 *
 * Kembali ke jam dan cuaca otomatis:
 * window.momoWorld.set({ hour: null, weather: null })
 */
export function setWorldOverride(
    options: { hour?: number | null; weather?: Weather | null },
) {
    if ('hour' in options) {
        hourOverride = options.hour ?? null
    }

    if ('weather' in options) {
        weatherOverride = options.weather ?? null
    }

    recompute()
}

if (typeof window !== 'undefined') {
    ; (
        window as unknown as {
            momoWorld?: { set: typeof setWorldOverride }
        }
    ).momoWorld = {
        set: setWorldOverride,
    }
}

import { NextRequest, NextResponse } from "next/server";
import { z } from "zod";
import { prisma } from "@/lib/prisma";
import { getCurrentUser } from "@/lib/auth";

export const dynamic = "force-dynamic";

const XP_POINTS = {
  creeper: 1,
  pet: 1,
  feed: 2,
  play: 5,
  orb: 2,
} as const;

const xpSchema = z.object({
  action: z.enum(["creeper", "pet", "feed", "play", "orb"]),
  requestId: z.string().uuid(),
}).strict();

const MAX_SCORE = 1_000_000;

class RateLimitError extends Error {
  constructor() {
    super("Batas aktivitas XP tercapai. Coba lagi setelah 60 detik.");
    this.name = "RateLimitError";
  }
}

function responseError(message: string, status: number) {
  return NextResponse.json(
    { success: false, message },
    { status, headers: { "Cache-Control": "no-store" } }
  );
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user) {
      return responseError("Silakan login terlebih dahulu.", 401);
    }

    if (request.headers.get("origin") !== request.nextUrl.origin) {
      return responseError("Origin request tidak diizinkan.", 403);
    }

    const contentType = request.headers.get("content-type") ?? "";

    if (!contentType.toLowerCase().startsWith("application/json")) {
      return responseError("Content-Type harus application/json.", 415);
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return responseError("Format JSON tidak valid.", 400);
    }

    const parsed = xpSchema.safeParse(body);

    if (!parsed.success) {
      return responseError("Data aktivitas XP tidak valid.", 400);
    }

    const { action, requestId } = parsed.data;
    const points = XP_POINTS[action];
    const now = new Date();

    const result = await prisma.$transaction(async (tx) => {
      await tx.momo_pets.upsert({
        where: { user_id: user.id },
        create: {
          user_id: user.id,
          name: "Momo",
          score: 0,
          energy: 80,
          happiness: 70,
          fullness: 65,
          asleep: false,
          skin: "salju",
          accessory: "none",
          animations: true,
          auto_move: true,
          chase: true,
          last_seen: now,
        },
        update: {},
      });

      // Catat transaksi lebih dulu agar request duplikat tidak
      // menghabiskan kuota rate limit.
      const inserted = await tx.momo_xp_events.createMany({
        data: [{
          user_id: user.id,
          request_id: requestId,
          action,
          points,
        }],
        skipDuplicates: true,
      });

      const currentPet = await tx.momo_pets.findUniqueOrThrow({
        where: { user_id: user.id },
      });

      if (inserted.count === 0) {
        return {
          score: currentPet.score,
          awarded: false,
          duplicate: true,
        };
      }

      // Maksimal 20 aktivitas dalam satu jendela 60 detik.
      // Operasi UPSERT ini atomik dan diserialisasi oleh row lock
      // PostgreSQL untuk setiap user_id.
      const rateLimit = await tx.$queryRaw<Array<{
        request_count: number;
      }>>`
        INSERT INTO public.momo_xp_rate_limits
          (user_id, window_started_at, request_count)
        VALUES
          (${user.id}, clock_timestamp(), 1)
        ON CONFLICT (user_id) DO UPDATE
        SET
          window_started_at = CASE
            WHEN public.momo_xp_rate_limits.window_started_at
              <= EXCLUDED.window_started_at - INTERVAL '60 seconds'
            THEN EXCLUDED.window_started_at
            ELSE public.momo_xp_rate_limits.window_started_at
          END,
          request_count = CASE
            WHEN public.momo_xp_rate_limits.window_started_at
              <= EXCLUDED.window_started_at - INTERVAL '60 seconds'
            THEN 1
            ELSE public.momo_xp_rate_limits.request_count + 1
          END
        WHERE
          public.momo_xp_rate_limits.window_started_at
            <= EXCLUDED.window_started_at - INTERVAL '60 seconds'
          OR public.momo_xp_rate_limits.request_count < 20
        RETURNING request_count
      `;

      if (rateLimit.length === 0) {
        // Exception akan membatalkan transaksi, termasuk pencatatan
        // event, sehingga request dapat dicoba kembali nanti.
        throw new RateLimitError();
      }

      const updated = await tx.momo_pets.updateMany({
        where: {
          user_id: user.id,
          score: { lte: MAX_SCORE - points },
        },
        data: {
          score: { increment: points },
          last_seen: now,
        },
      });

      const pet = await tx.momo_pets.findUniqueOrThrow({
        where: { user_id: user.id },
      });

      return {
        score: pet.score,
        awarded: updated.count === 1,
        duplicate: false,
      };
    });

    return NextResponse.json(
      {
        success: true,
        data: result,
      },
      { headers: { "Cache-Control": "no-store" } }
    );
  } catch (error) {
    if (error instanceof RateLimitError) {
      return NextResponse.json(
        {
          success: false,
          message: error.message,
        },
        {
          status: 429,
          headers: {
            "Cache-Control": "no-store",
            "Retry-After": "60",
          },
        }
      );
    }

    console.error("POST /api/momo/xp:", error);
    return responseError("Gagal menambahkan XP Momo.", 500);
  }
}

import { NextRequest, NextResponse } from "next/server";
import { z } from "zod";
import { prisma } from "@/lib/prisma";
import { getCurrentUser } from "@/lib/auth";

const moodLabels = [
  "Lelah/Down",
  "Cemas",
  "Biasa aja",
  "Tenang",
  "Semangat!",
] as const;

const checkinSchema = z.object({
  mood: z.enum(moodLabels),
  energyLevel: z.number().int().min(1).max(5),
  sleepHours: z.number()
    .min(0)
    .max(24)
    .refine((value) => Number.isInteger(value * 2), {
      message: "Durasi tidur harus kelipatan 0,5 jam.",
    }),
  activities: z.array(z.enum([
    "Kuliah",
    "Proyek Kelompok",
    "Lab/Praktikum",
  ])).max(3),
  otherActivity: z.string().trim().max(255),
}).strict();

function errorResponse(message: string, status: number) {
  return NextResponse.json(
    { success: false, message },
    {
      status,
      headers: { "Cache-Control": "no-store" },
    }
  );
}

function getTodayWIB() {
  const dateString = new Intl.DateTimeFormat("en-CA", {
    timeZone: "Asia/Jakarta",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).format(new Date());

  return new Date(`${dateString}T00:00:00.000Z`);
}

export async function POST(request: NextRequest) {
  const user = await getCurrentUser();

  if (!user) {
    return errorResponse("Silakan login terlebih dahulu.", 401);
  }

  if (request.headers.get("origin") !== request.nextUrl.origin) {
    return errorResponse("Origin request tidak diizinkan.", 403);
  }

  const contentType = request.headers.get("content-type") ?? "";

  if (!contentType.toLowerCase().startsWith("application/json")) {
    return errorResponse("Content-Type harus application/json.", 415);
  }

  const rawBody = await request.text();

  if (Buffer.byteLength(rawBody, "utf8") > 10_000) {
    return errorResponse("Ukuran request terlalu besar.", 413);
  }

  let body: unknown;

  try {
    body = JSON.parse(rawBody);
  } catch {
    return errorResponse("Format JSON tidak valid.", 400);
  }

  const parsed = checkinSchema.safeParse(body);

  if (!parsed.success) {
    return NextResponse.json(
      {
        success: false,
        message: "Data Check-in tidak valid.",
        errors: z.treeifyError(parsed.error),
      },
      {
        status: 400,
        headers: { "Cache-Control": "no-store" },
      }
    );
  }

  const { mood, energyLevel, sleepHours, activities, otherActivity } =
    parsed.data;

  const moodNumber = moodLabels.indexOf(mood) + 1;
  const today = getTodayWIB();
  const now = new Date();

  const activityData = JSON.stringify({
    activities,
    otherActivity,
  });

  try {
    const result = await prisma.$transaction(async (tx) => {
      const checkin = await tx.daily_check_ins.upsert({
        where: {
          user_id_checkin_date: {
            user_id: user.id,
            checkin_date: today,
          },
        },
        create: {
          user_id: user.id,
          checkin_date: today,
          mood: moodNumber,
          energy: energyLevel,
          sleep_hours: sleepHours,
          activity: activityData,
          created_at: now,
          updated_at: now,
        },
        update: {
          mood: moodNumber,
          energy: energyLevel,
          sleep_hours: sleepHours,
          activity: activityData,
          updated_at: now,
        },
      });

      // Pastikan data Momo sudah tersedia untuk akun ini.
      const momo = await tx.momo_pets.upsert({
        where: { user_id: user.id },
        create: {
          user_id: user.id,
          name: "Momo",
          score: 0,
          energy: 80,
          happiness: 70,
          fullness: 65,
          asleep: false,
          skin: "salju",
          accessory: "none",
          animations: true,
          auto_move: true,
          chase: true,
          last_seen: now,
        },
        update: {},
      });

      // Kondisi tanggal membuat hadiah hanya bisa diklaim sekali sehari.
      const reward = await tx.momo_pets.updateMany({
        where: {
          user_id: user.id,
          OR: [
            { last_checkin: null },
            { last_checkin: { lt: today } },
          ],
        },
        data: {
          score: { increment: 10 },
          happiness: Math.min(100, momo.happiness + 20),
          last_checkin: today,
          last_seen: now,
        },
      });

      return {
        checkin,
        momoRewardAwarded: reward.count === 1,
      };
    });

    return NextResponse.json(
      {
        success: true,
        message: "Check-in hari ini berhasil disimpan.",
        momoRewardAwarded: result.momoRewardAwarded,
        data: {
          id: result.checkin.id.toString(),
          date: result.checkin.checkin_date.toISOString().slice(0, 10),
          mood,
          energyLevel: result.checkin.energy,
          sleepHours: Number(result.checkin.sleep_hours),
          activities,
          otherActivity,
        },
      },
      {
        headers: { "Cache-Control": "no-store" },
      }
    );
  } catch (error) {
    console.error("POST /api/checkin:", error);
    return errorResponse("Check-in gagal disimpan. Silakan coba lagi.", 500);
  }
}

  otherActivity: string

type ApiResponse = {

  message?: string
  momoRewardAwarded?: boolean
  data?: CheckinData | null

const moods: Mood[] = [
  { image: '/images/moods/sedih.svg', label: 'Lelah/Down' },
  { image: '/images/moods/cemas.svg', label: 'Cemas' },
  { image: '/images/moods/biasa aja.svg', label: 'Biasa aja' },
  { image: '/images/moods/tenang.svg', label: 'Tenang' },
  { image: '/images/moods/semangat.svg', label: 'Semangat!' },
]

const activityOptions = [
  'Kuliah',
  'Proyek Kelompok',
  'Lab/Praktikum',
]

const energyLabels = [
  '',
  'Sangat baterai lemah',
  'Baterai lemah',
  'Cukup',
  'Berenergi',
  'Sangat berenergi',
]

export default function CheckinForm() {
  const router = useRouter()

  const [selectedMood, setSelectedMood] = useState<string | null>(null)
  const [energyLevel, setEnergyLevel] = useState(3)
  const [sleepHours, setSleepHours] = useState('')
  const [activities, setActivities] = useState<string[]>([])
  const [otherActivity, setOtherActivity] = useState('')
  const [message, setMessage] = useState('')
  const [showSuccess, setShowSuccess] = useState(false)
  const [isLoading, setIsLoading] = useState(true)
  const [isSaving, setIsSaving] = useState(false)

  const loadTodayCheckin = useCallback(async (signal: AbortSignal) => {
    setIsLoading(true)

    try {
      const response = await fetch('/api/checkin/today', {
        method: 'GET',
        cache: 'no-store',
        signal,
      })

      const result = (await response.json()) as ApiResponse

      if (!response.ok || !result.success) {
        if (response.status === 401) {
          setMessage(
            'Silakan login terlebih dahulu untuk menggunakan Check-in.'
          )
        } else {
          setMessage(result.message ?? 'Data Check-in gagal dimuat.')
        }
        return
      }

      if (!result.data) {
        return
      }

      setSelectedMood(result.data.mood)
      setEnergyLevel(result.data.energyLevel)
      setSleepHours(String(result.data.sleepHours))
      setActivities(result.data.activities ?? [])
      setOtherActivity(result.data.otherActivity ?? '')
    } catch (error) {
      if (error instanceof Error && error.name === 'AbortError') {
        return
      }

      setMessage(
        'Tidak dapat memuat Check-in. Periksa koneksi lalu coba lagi.'
      )
    } finally {
      if (!signal.aborted) {
        setIsLoading(false)
      }
    }
  }, [])

  useEffect(() => {
    const controller = new AbortController()
    void loadTodayCheckin(controller.signal)

    return () => controller.abort()
  }, [loadTodayCheckin])

  function toggleActivity(activity: string) {
    setActivities((previous) => {
      if (previous.includes(activity)) {
        return previous.filter((item) => item !== activity)
      }

      if (previous.length >= 3) {
        return previous
      }

      return [...previous, activity]
    })

    setMessage((previous) =>
      activities.length >= 3 && !activities.includes(activity)
        ? 'Pilih maksimal 3 aktivitas utama.'
        : ''
    )
  }

  async function handleSave() {
    if (isSaving || isLoading) return

    if (!selectedMood) {
      setMessage('Pilih suasana hatimu terlebih dahulu, ya 💚')
      return
    }

    if (sleepHours.trim() === '') {
      setMessage('Masukkan durasi tidurmu terlebih dahulu.')
      return
    }

    const hours = Number(sleepHours)

    if (!Number.isFinite(hours) || hours < 0 || hours > 24) {
      setMessage('Jam istirahat harus antara 0 sampai 24 jam, ya.')
      return
    }

    if (!Number.isInteger(hours * 2)) {
      setMessage('Durasi tidur harus dalam kelipatan 0,5 jam.')
      return
    }

    if (activities.length > 3) {
      setMessage('Pilih maksimal 3 aktivitas utama.')
      return
    }

    if (otherActivity.trim().length > 255) {
      setMessage('Aktivitas lainnya maksimal 255 karakter.')
      return
    }

    setIsSaving(true)
    setMessage('')

    try {
      const response = await fetch('/api/checkin', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          mood: selectedMood,
          energyLevel,
          sleepHours: hours,
          activities,
          otherActivity: otherActivity.trim(),
        }),
      })

      const result = (await response.json()) as ApiResponse

      if (!response.ok || !result.success) {
        if (response.status === 401) {
          setMessage('Sesi login tidak ditemukan. Silakan login kembali.')
        } else {
          setMessage(
            result.message ?? 'Check-in gagal disimpan. Coba lagi.'
          )
        }
        return
      }

      // Hadiah Momo sudah diberikan langsung oleh API dan disimpan ke database.
      setShowSuccess(true)
    } catch {
      setMessage(
        'Tidak dapat terhubung ke server. Periksa koneksi lalu coba lagi.'
      )
    } finally {
      setIsSaving(false)
    }
  }

  return (
    <>
      <div className="checkin-demo-card">
        <div className="checkin-moods" aria-busy={isLoading}>
          {moods.map((mood) => (
            <button
              key={mood.label}
              type="button"
              className={`mood-option ${selectedMood === mood.label ? 'selected' : ''
                }`}
              onClick={() => {
                setSelectedMood(mood.label)
                setMessage('')
              }}
              aria-pressed={selectedMood === mood.label}
              disabled={isLoading || isSaving}
            >
              <img
                src={mood.image}
                alt=""
                className="mood-image"
                aria-hidden="true"
              />

              <small>{mood.label}</small>
            </button>
          ))}
        </div>

        <div className="checkin-energy-panel">
          <div className="checkin-energy-header">
            <label htmlFor="daily-energy-level">
              TINGKAT ENERGI BATERAI DIRI
            </label>

            <span className="checkin-energy-badge">
              {energyLevel}/5 ({energyLabels[energyLevel]})
            </span>
          </div>

          <input
            id="daily-energy-level"
            className="checkin-energy-slider"
            type="range"
            min={1}
            max={5}
            step={1}
            value={energyLevel}
            onChange={(event) =>
              setEnergyLevel(Number(event.target.value))
            }
            aria-valuetext={energyLabels[energyLevel]}
            disabled={isLoading || isSaving}
          />
        </div>

        <div className="checkin-bottom-grid">
          <div className="checkin-info-panel">
            <label htmlFor="daily-sleep-hours">
              ISTIRAHAT SEMALAM
            </label>

            <div className="checkin-sleep-input">
              <input
                id="daily-sleep-hours"
                type="number"
                min="0"
                max="24"
                step="0.5"
                placeholder="Contoh: 7"
                value={sleepHours}
                onChange={(event) => {
                  setSleepHours(event.target.value)
                  setMessage('')
                }}
                disabled={isLoading || isSaving}
              />

              <span>Jam</span>
            </div>
          </div>

          <div className="checkin-info-panel">
            <p className="checkin-panel-title">
              AKTIVITAS UTAMA
            </p>

            <div className="checkin-activities">
              {activityOptions.map((activity) => (
                <button
                  key={activity}
                  type="button"
                  className={
                    activities.includes(activity) ? 'selected' : ''
                  }
                  onClick={() => toggleActivity(activity)}
                  aria-pressed={activities.includes(activity)}
                  disabled={isLoading || isSaving}
                >
                  {activity}
                </button>
              ))}
            </div>

            <div className="checkin-other-activity">
              <label
                htmlFor="daily-other-activity"
                className="sr-only"
              >
                Aktivitas lainnya
              </label>

              <input
                id="daily-other-activity"
                type="text"
                maxLength={255}
                placeholder="Tulis aktivitas lainnya..."
                value={otherActivity}
                onChange={(event) => {
                  setOtherActivity(event.target.value)
                  setMessage('')
                }}
                disabled={isLoading || isSaving}
              />
            </div>
          </div>
        </div>

        <button
          className="checkin-save"
          type="button"
          onClick={handleSave}
          disabled={isLoading || isSaving}
        >
          {isLoading
            ? 'Memuat Check-in...'
            : isSaving
              ? 'Menyimpan...'
              : 'Simpan Check-in Hari Ini'}
        </button>

        {message && (
          <p className="checkin-message" role="status">
            {message}
          </p>
        )}
      </div>

      {showSuccess && (
        <div className="daily-checkin-overlay" role="presentation">
          <div
            className="daily-checkin-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="daily-checkin-success-title"
            aria-describedby="daily-checkin-success-description"
          >
            <div
              className="daily-checkin-success-icon"
              aria-hidden="true"
            >
              ✓
            </div>

            <h2 id="daily-checkin-success-title">
              Berhasil!
            </h2>

            <p id="daily-checkin-success-description">
              Check-in harian berhasil disimpan ke database.
            </p>

            <button
              type="button"
              onClick={() => {
                setShowSuccess(false)
                router.push('/dashboard')
                router.refresh()
              }}
              autoFocus
            >
              Oke
            </button>
          </div>
        </div>
      )}
    </>
  )
}
